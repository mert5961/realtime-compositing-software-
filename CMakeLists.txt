cmake_minimum_required(VERSION 3.20)
project(RealtimeCompositor VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(compositor_core src/core.cpp)
target_include_directories(compositor_core PUBLIC include)
target_link_libraries(compositor_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(compositor_core PUBLIC /W4 /permissive- /EHsc)
else()
    target_compile_options(compositor_core PUBLIC -Wall -Wextra -Wpedantic)
endif()

option(COMPOSITOR_BUILD_SPOUT "Build Windows Blender texture-sharing prototype" OFF)
set(SPOUT_SDK_DIR "${CMAKE_SOURCE_DIR}/build/deps/Spout2" CACHE PATH "Spout2 SDK source directory")
if(COMPOSITOR_BUILD_SPOUT AND WIN32)
    set(SPOUT_BUILD_CMT OFF CACHE BOOL "" FORCE)
    set(SPOUT_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SPOUT_BUILD_SPOUTDX ON CACHE BOOL "" FORCE)
    set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
    add_subdirectory("${SPOUT_SDK_DIR}" "spout-sdk" EXCLUDE_FROM_ALL)
    # Frame counters are required for diagnostics. Enable them only in these
    # binaries; do not change the user's global Spout registry preferences.
    set(_spout_count "${SPOUT_SDK_DIR}/SPOUTSDK/SpoutGL/SpoutFrameCount.cpp")
    file(READ "${_spout_count}" _spout_count_source)
    set(_count_hook "m_bFrameCount = (dwFrame == 1);")
    string(FIND "${_spout_count_source}" "${_count_hook}" _count_position)
    if(_count_position EQUAL -1)
        message(FATAL_ERROR "Pinned Spout frame-count hook changed")
    endif()
    string(REPLACE "m_bFrameCount = false; // default not set" "m_bFrameCount = true; // Application-local diagnostics" _spout_count_source "${_spout_count_source}")
    string(REPLACE "${_count_hook}" "m_bFrameCount = true; // Application-local diagnostics" _spout_count_source "${_spout_count_source}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/spout-frame-count.cpp" "${_spout_count_source}")
    foreach(_spout_target Spout_static SpoutDX_static)
        get_target_property(_spout_sources ${_spout_target} SOURCES)
        list(FILTER _spout_sources EXCLUDE REGEX "SpoutFrameCount\\.cpp$")
        set_property(TARGET ${_spout_target} PROPERTY SOURCES "${_spout_sources}")
        target_sources(${_spout_target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/spout-frame-count.cpp")
    endforeach()
    add_executable(blender_texture_test WIN32 src/blender_texture_test.cpp)
    target_compile_definitions(blender_texture_test PRIVATE NOMINMAX)
    target_link_libraries(blender_texture_test PRIVATE SpoutDX_static d3dcompiler)
    add_library(blender_spout_bridge SHARED src/blender_spout_bridge.cpp)
    target_compile_definitions(blender_spout_bridge PRIVATE NOMINMAX)
    target_link_libraries(blender_spout_bridge PRIVATE Spout_static)
    add_custom_command(TARGET blender_spout_bridge POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SPOUT_SDK_DIR}/LICENSE"
            "$<TARGET_FILE_DIR:blender_spout_bridge>/Spout-LICENSE.txt")
endif()
