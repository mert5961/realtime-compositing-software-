cmake_minimum_required(VERSION 3.20)
project(RealtimeCompositor VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(compositor_core src/core.cpp)
target_include_directories(compositor_core PUBLIC include)
target_link_libraries(compositor_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(compositor_core PUBLIC /W4 /permissive- /EHsc)
else()
    target_compile_options(compositor_core PUBLIC -Wall -Wextra -Wpedantic)
endif()

option(COMPOSITOR_BUILD_AJA "Build Windows SDI test UI with AJA SDK" OFF)
set(AJA_SDK_DIR "${CMAKE_SOURCE_DIR}/build/deps/libajantv2" CACHE PATH "AJA NTV2 SDK source directory")
if(COMPOSITOR_BUILD_AJA AND WIN32)
    set(AJANTV2_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_TESTS ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_TOOLS ON CACHE BOOL "" FORCE)
    set(AJANTV2_DISABLE_DRIVER ON CACHE BOOL "" FORCE)
    add_subdirectory("${AJA_SDK_DIR}" "aja-sdk" EXCLUDE_FROM_ALL)
    include(cmake/AjaWindowsCompatibility.cmake)
    include(cmake/AjaProfiling.cmake)
    add_executable(sdi_test src/sdi_test_win.cpp
        "${CMAKE_CURRENT_BINARY_DIR}/aja-burn.cpp"
        "${AJA_SDK_DIR}/demos/ntv2democommon.cpp")
    target_include_directories(sdi_test PRIVATE "${CMAKE_SOURCE_DIR}/src" "${AJA_SDK_DIR}" "${AJA_SDK_DIR}/demos"
        "${AJA_SDK_DIR}/demos/ntv2burn" "${AJA_SDK_DIR}/ajantv2/includes")
    target_compile_definitions(sdi_test PRIVATE NOMINMAX)
    target_link_libraries(sdi_test PRIVATE ajantv2 user32 gdi32 ws2_32)
endif()

add_executable(compositor_demo src/main.cpp)
target_link_libraries(compositor_demo PRIVATE compositor_core)

include(CTest)
if(BUILD_TESTING)
    if(COMPOSITOR_BUILD_AJA AND WIN32)
        add_executable(sdi_profile_tests tests/sdi_profile_tests.cpp)
        target_include_directories(sdi_profile_tests PRIVATE src)
        target_link_libraries(sdi_profile_tests PRIVATE Threads::Threads)
        add_test(NAME sdi_profile_metrics COMMAND sdi_profile_tests)
    endif()
    add_executable(compositor_tests tests/core_tests.cpp)
    target_link_libraries(compositor_tests PRIVATE compositor_core)
    add_test(NAME compositor_core_tests COMMAND compositor_tests)
    add_test(NAME compositor_demo_scenario COMMAND compositor_demo "${CMAKE_CURRENT_BINARY_DIR}/demo-output")
endif()
